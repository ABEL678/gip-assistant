from enum import Enum


class DocumentKind(str, Enum):
    """Вид документации (ГОСТ Р 21.101-2026, 4.1, 4.2)."""
    PD = "ПД"  # проектная документация
    RD = "РД"  # рабочая документация

    TCH = "ТЧ"  # текстовая часть
    GCH = "ГЧ"  # графическая часть


class LifecycleStatus(str, Enum):
    """Жизненный цикл документа (ГОСТ Р 21.101-2026, 7.1-7.5)."""
    DRAFT = "DRAFT"
    IN_REVIEW = "IN_REVIEW"
    NORMOCONTROL = "NORMOCONTROL"
    APPROVED = "APPROVED"
    ARCHIVED = "ARCHIVED"
    ANNULLED = "ANNULLED"
    REPLACED = "REPLACED"


class DocumentLinkType(str, Enum):
    """Тип связи между документами (ГОСТ Р 21.101-2026, 4.3.3)."""
    APPENDIX = "appendix"       # прилагаемый
    REFERENCE = "reference"     # ссылочный
    REPLACES = "replaces"       # заменяет
    VERSION_OF = "version_of"   # версия

## 3. SQLAlchemy-модели

**Важно:** НЕ используем TimestampMixin. Каждая модель самостоятельно управляет `created_at`/`updated_at`.

### 3.1. DocumentTypeModel

**Файл:** `backend/src/adapters/db/models/document_type.py`

```python
import uuid
from datetime import datetime

from sqlalchemy import String, Text
from sqlalchemy.orm import Mapped, mapped_column

from backend.src.adapters.db.base import Base


class DocumentTypeModel(Base):
    __tablename__ = "document_types"

    id: Mapped[uuid.UUID] = mapped_column(
        primary_key=True, default=uuid.uuid4
    )
    code: Mapped[str] = mapped_column(String(50), nullable=False, unique=True)
    name: Mapped[str] = mapped_column(String(200), nullable=False)
    description: Mapped[str | None] = mapped_column(Text, nullable=True)
    gost_code: Mapped[str | None] = mapped_column(String(50), nullable=True)
    created_at: Mapped[datetime] = mapped_column(
        default=datetime.now, nullable=False
    )
```

### 3.2. DocumentModel

**Файл:** `backend/src/adapters/db/models/document.py`

```python
import uuid
from datetime import datetime

from sqlalchemy import (
    ForeignKey, String, Text,
    Index,
)
from sqlalchemy.orm import Mapped, mapped_column

from backend.src.adapters.db.base import Base
from backend.src.domain.value_objects.document import (
    DocumentKind, DocPartType, LifecycleStatus,
)


class DocumentModel(Base):
    __tablename__ = "documents"
    __table_args__ = (
        Index("idx_documents_project", "project_id"),
        Index("idx_documents_entity", "entity_id"),
        Index("idx_documents_task", "task_id"),
        Index("idx_documents_guid", "guid"),
        Index("idx_documents_type", "document_type_id"),
        Index("idx_documents_status", "lifecycle_status"),
    )

    id: Mapped[uuid.UUID] = mapped_column(
        primary_key=True, default=uuid.uuid4
    )
    guid: Mapped[uuid.UUID] = mapped_column(
        nullable=False, unique=True
    )
    project_id: Mapped[uuid.UUID] = mapped_column(
        ForeignKey("projects.id", ondelete="CASCADE"), nullable=False
    )
    entity_id: Mapped[uuid.UUID | None] = mapped_column(
        ForeignKey("entities.id", ondelete="SET NULL"), nullable=True
    )
    task_id: Mapped[uuid.UUID | None] = mapped_column(
        ForeignKey("tasks.id", ondelete="SET NULL"), nullable=True
    )

    # Обозначение (ГОСТ Р 21.101-2026, 4.1, 4.2)
    cipher: Mapped[str] = mapped_column(String(500), nullable=False, default="")
    base_designation: Mapped[str | None] = mapped_column(String(200), nullable=True)
    section_code: Mapped[str | None] = mapped_column(String(50), nullable=True)
    document_kind: Mapped[DocumentKind] = mapped_column(
        String(10), nullable=False, default=DocumentKind.PD
    )
    mark: Mapped[str | None] = mapped_column(String(50), nullable=True)
    doc_part_type: Mapped[DocPartType | None] = mapped_column(
        String(10), nullable=True
    )

    # Наименование
    document_name: Mapped[str] = mapped_column(String(500), nullable=False, default="")
    document_type_id: Mapped[uuid.UUID] = mapped_column(
        ForeignKey("document_types.id"), nullable=False
    )

    # Версионность
    current_version_id: Mapped[uuid.UUID | None] = mapped_column(nullable=True)
    version_parent_id: Mapped[uuid.UUID | None] = mapped_column(
        ForeignKey("documents.id"), nullable=True
    )
    replaces_id: Mapped[uuid.UUID | None] = mapped_column(
        ForeignKey("documents.id"), nullable=True
    )
    replaced_by_id: Mapped[uuid.UUID | None] = mapped_column(
        ForeignKey("documents.id"), nullable=True
    )

    # Жизненный цикл
    lifecycle_status: Mapped[LifecycleStatus] = mapped_column(
        String(20), nullable=False, default=LifecycleStatus.DRAFT
    )

    # Подпись (метаданные)
    signed_by: Mapped[uuid.UUID | None] = mapped_column(
        ForeignKey("users.id"), nullable=True
    )
    signed_at: Mapped[datetime | None] = mapped_column(nullable=True)
    signature_hash: Mapped[str | None] = mapped_column(String(255), nullable=True)
    signature_status: Mapped[str | None] = mapped_column(String(50), nullable=True)

    # Служебные
    created_by: Mapped[uuid.UUID | None] = mapped_column(
        ForeignKey("users.id"), nullable=True
    )
    created_at: Mapped[datetime] = mapped_column(default=datetime.now, nullable=False)
    updated_at: Mapped[datetime] = mapped_column(default=datetime.now, nullable=False)
```

### 3.3. DocumentVersionModel

**Файл:** `backend/src/adapters/db/models/document_version.py`

```python
import uuid
from datetime import datetime

from sqlalchemy import (
    ForeignKey, String, BigInteger,
    Boolean, Index,
)
from sqlalchemy.orm import Mapped, mapped_column

from backend.src.adapters.db.base import Base


class DocumentVersionModel(Base):
    __tablename__ = "document_versions"
    __table_args__ = (
        Index("idx_document_versions_document", "document_id"),
        Index(
            "idx_document_versions_current",
            "document_id", "is_current",
            postgresql_where="is_current = TRUE",
        ),
    )

    id: Mapped[uuid.UUID] = mapped_column(
        primary_key=True, default=uuid.uuid4
    )
    document_id: Mapped[uuid.UUID] = mapped_column(
        ForeignKey("documents.id", ondelete="CASCADE"), nullable=False
    )
    version: Mapped[str] = mapped_column(String(20), nullable=False)
    change_number: Mapped[int | None] = mapped_column(nullable=True)
    file_path: Mapped[str] = mapped_column(String(1000), nullable=False)
    file_name: Mapped[str] = mapped_column(String(500), nullable=False)
    hash_sha256: Mapped[str] = mapped_column(String(64), nullable=False)
    format_original: Mapped[str] = mapped_column(String(20), nullable=False)
    format_archive: Mapped[str | None] = mapped_column(String(20), nullable=True)
    size_bytes: Mapped[int] = mapped_column(BigInteger, nullable=False)
    is_current: Mapped[bool] = mapped_column(Boolean, nullable=False, default=False)
    uploaded_by: Mapped[uuid.UUID | None] = mapped_column(
        ForeignKey("users.id"), nullable=True
    )
    uploaded_at: Mapped[datetime] = mapped_column(
        default=datetime.now, nullable=False
    )
```

### 3.4. DocumentLinkModel

**Файл:** `backend/src/adapters/db/models/document_link.py`

```python
import uuid
from datetime import datetime

from sqlalchemy import (
    ForeignKey, String,
    Index, UniqueConstraint,
)
from sqlalchemy.orm import Mapped, mapped_column

from backend.src.adapters.db.base import Base


class DocumentLinkModel(Base):
    __tablename__ = "document_links"
    __table_args__ = (
        UniqueConstraint("source_document_id", "target_document_id", "link_type"),
        Index("idx_document_links_source", "source_document_id"),
        Index("idx_document_links_target", "target_document_id"),
    )

    id: Mapped[uuid.UUID] = mapped_column(
        primary_key=True, default=uuid.uuid4
    )
    source_document_id: Mapped[uuid.UUID] = mapped_column(
        ForeignKey("documents.id", ondelete="CASCADE"), nullable=False
    )
    target_document_id: Mapped[uuid.UUID] = mapped_column(
        ForeignKey("documents.id", ondelete="CASCADE"), nullable=False
    )
    link_type: Mapped[str] = mapped_column(String(30), nullable=False)
    created_at: Mapped[datetime] = mapped_column(
        default=datetime.now, nullable=False
    )
```
